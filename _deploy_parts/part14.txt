const JA_PART2=["までは","からは","にとって","として","について","のために","ながら","けれど"];
const JA_PART1=["を","に","が","は","で","と","も","へ","や","の","から","まで","より","ば"];
const _KJ=/[一-鿿々゠-ヿ]/;      // 한자·가타카나 = 새 문절 시작 신호
// 일본어 문절(어절) 분할 — 형태소분석기 없이 보수적으로:
//  ① 、。！？ 뒤에서는 항상 끊는다
//  ② 조사 뒤는 "다음 글자가 한자/가타카나일 때만" 끊는다 (まいにち·ながら 오분할 방지)
function jaChunks(t){
  const s=String(t).replace(/\s+/g,"");
  const out=[]; let cur="";
  for(let i=0;i<s.length;i++){
    cur+=s[i];
    const ch=s[i], nx=s[i+1]||"";
    if("、。！？".includes(ch)){ if(nx) { out.push(cur); cur=""; } continue; }
    if(cur.length<2 || !nx) continue;
    if(!_KJ.test(nx)) continue;                        // 다음이 히라가나면 단어 중간일 가능성 ↑
    let hit=false;
    for(const q of ["から","まで","より"]) if(s.slice(i-q.length+1,i+1)===q && cur.length>q.length){ hit=true; break; }
    if(!hit && "をにがはでともへやの".includes(ch) && cur.length>=2) hit=true;
    if(hit){ out.push(cur); cur=""; }
  }
  if(cur) out.push(cur);
  // 1글자 조각 병합
  let m=[];
  for(const c of out){
    if(m.length && (c.length<=1 || m[m.length-1].length<=1)) m[m.length-1]+=c;
    else m.push(c);
  }
  // 8조각 이상이면 가장 짧은 이웃끼리 합쳐 7조각 이하로
  while(m.length>7){
    let bi=0, bs=1e9;
    for(let i=0;i<m.length-1;i++){ const v=m[i].length+m[i+1].length; if(v<bs){bs=v;bi=i;} }
    m.splice(bi,2,m[bi]+m[bi+1]);
  }
  return m;
}
