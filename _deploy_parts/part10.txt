// ══════════ 동사 활용 엔진 + 연습 모드 (전부 클라이언트) ══════════
const CJ_U="うくぐすつぬぶむる";
const CJ_I={"う":"い","く":"き","ぐ":"ぎ","す":"し","つ":"ち","ぬ":"に","ぶ":"び","む":"み","る":"り"};
const CJ_A={"う":"わ","く":"か","ぐ":"が","す":"さ","つ":"た","ぬ":"な","ぶ":"ば","む":"ま","る":"ら"};
const CJ_TE={"う":"って","つ":"って","る":"って","ぬ":"んで","ぶ":"んで","む":"んで","く":"いて","ぐ":"いで","す":"して"};
const CJ_TA={"って":"った","んで":"んだ","いて":"いた","いで":"いだ","して":"した"};
const CJ_FORMS=[{k:"masu",ko:"ます형",jp:"ます形"},{k:"te",ko:"て형",jp:"て形"},
                {k:"nai",ko:"ない형",jp:"ない形"},{k:"ta",ko:"た형",jp:"た形"}];
// grp: 1|2|3 , rd: 사전형 히라가나, fr: 사전형 표기(한자 가능)
function conj(fr, rd, grp, form){
  const cut=(n,tail)=>({f:fr.slice(0,-n)+tail, r:rd.slice(0,-n)+tail});
  if(grp===3){
    const n = rd.endsWith("する")?2:2;                       // する / くる 둘 다 2글자
    const isKuru = rd.endsWith("くる");
    const tails = isKuru
      ? {masu:["ます","きます"],te:["て","きて"],nai:["ない","こない"],ta:["た","きた"]}
      : {masu:["します","します"],te:["して","して"],nai:["しない","しない"],ta:["した","した"]};
    const [ft,rt]=tails[form];
    if(isKuru){                                              // 来る → 来ます(きます)
      return {f:fr.slice(0,-1)+ft, r:rd.slice(0,-2)+rt};
    }
    return {f:fr.slice(0,-n)+ft, r:rd.slice(0,-n)+rt};
  }
  if(grp===2){
    const t={masu:"ます",te:"て",nai:"ない",ta:"た"}[form];
    return cut(1,t);
  }
  const last=rd.slice(-1);
  if(form==="masu") return cut(1, CJ_I[last]+"ます");
  if(form==="nai")  return cut(1, CJ_A[last]+"ない");
  let te = CJ_TE[last];
  if(rd==="いく") te="って";                                  // 行く 예외
  if(form==="te")  return cut(1, te);
  return cut(1, CJ_TA[te]);
}
// 그럴듯한 오답 생성 (다른 그룹 규칙 적용 / 어미 혼동)
function cjWrong(fr, rd, grp, form, correct){
  const key=o=>o.f+"|"+o.r;
  const seen=new Set([key(correct)]); const out=[];
  // 표기(f)가 정답과 같으면 화면상 중복 → 제외
  const add=o=>{ if(o && o.r && o.f && o.f!==correct.f && !seen.has(key(o))){ seen.add(key(o)); out.push(o); } };
  const safe=(f,r,g,fm)=>{ try{ return conj(f,r,g,fm); }catch(e){ return null; } };
  // (a) 다른 활용그룹 규칙을 잘못 적용한 형태
  [1,2,3].forEach(g=>{ if(g!==grp) add(safe(fr,rd,g,form)); });
  // (b) 같은 동사의 다른 활용형 (형태 혼동)
  CJ_FORMS.forEach(f=>{ if(f.k!==form) add(safe(fr,rd,grp,f.k)); });
  // (c) 1류: 어미를 다른 う단으로 착각했을 때의 형태 (て/た형 함정)
  if(grp===1){
    const stemF=fr.slice(0,-1), stemR=rd.slice(0,-1);
    "うくぐすつぬぶむる".split("").forEach(k=>{
      if(k===rd.slice(-1)) return;
      const o=safe(stemF+k, stemR+k, 1, form);
      if(o) add({f:stemF+o.f.slice(stemF.length), r:stemR+o.r.slice(stemR.length)});
    });
  }
  return out;
}
