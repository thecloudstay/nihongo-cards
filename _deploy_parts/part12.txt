async function rdPlayAll(){
  if(_rdPlaying){ rdStopPlay(); return; }         // ⚠ 예전엔 다시 누르면 두 번째 재생이 겹쳐 소리가 끊겼음

    (rdCont && bk.chapters.length>1) ? `⏸ 멈춤 (${rdChapCur+1}/${bk.chapters.length}장)` : "⏸ 멈춤";
  label();

    const c=bk.chapters[rdChapCur];
    const els=[...$("rrBody").querySelectorAll(".rd-s")];
    for(let i=0;i<c.s.length;i++){
      if(my!==_rdPlayTok){ return; }

      els[i].classList.add("cur");

      els[i].scrollIntoView({block:"center",behavior:"smooth"});

      if(my!==_rdPlayTok) return;
      await new Promise(r=>setTimeout(r,180));
    }
    // ── 장 끝 ──
    if(!rdCont || rdChapCur+1 >= bk.chapters.length) break;
    if(my!==_rdPlayTok) return;
    _rdAuto=true;                 // openChapter 안의 rdStopPlay 가 재생을 끊지 않도록
    openChapter(rdChapCur+1);
    _rdAuto=false;

    await new Promise(r=>setTimeout(r,700));   // 새 장이 그려질 때까지 잠깐 쉼
    if(my!==_rdPlayTok) return;
  }
  const done = rdChapCur+1 >= bk.chapters.length;
  rdStopPlay();
  rdToTop();                      // 다 읽었으면 맨 위로 — 다시 읽거나 시험 보러 가기 편하게

// ⚠ "뜻 정보 없음"이 너무 자주 뜨던 원인: 표기(w) 완전일치만 찾았기 때문.
//    한 번만 색인(Map)을 만들어 두고 표기·읽기·한자 낱글자까지 단계적으로 훑는다.
let _wpW="", _rdIdx=null;
function rdBuildIndex(){
  if(_rdIdx) return _rdIdx;
  const byW=new Map(), byR=new Map(), byK=new Map();   // 표기 / 읽기 / 한자1글자
  const put=(m,k,v)=>{ if(k&&v&&!m.has(k)) m.set(k,v); };
  if(BOOKS) for(const b of BOOKS) for(const ch of b.chapters) for(const x of (ch.words||[])){
    put(byW,x.w,x.ko); put(byR,x.r,x.ko);
  }
  for(const k in DATA) for(const card of (DATA[k]||[])){
    const ko=card.ko; if(!ko || card.sent) continue;      // 문장카드는 단어 뜻이 아니므로 제외
    put(byW,card.front,ko);
    // 비(非)카나 카드는 romaji 필드에 "읽기(가나)"가 들어 있다 (rowToCard 참고)
    if(card.romaji && /[ぁ-ヿ]/.test(card.romaji)) String(card.romaji).split("/").forEach(x=>put(byR,x.trim(),ko));
    if(card.front && card.front.length===1) put(byK,card.front,ko);
  }
  return (_rdIdx={byW,byR,byK});
}
const rdTrim = s => String(s||"").replace(/[はがをにへでとものやかなねよぞぜさこそあうえおっ]+$/,"");
function rdLookup(w,r){
  const I=rdBuildIndex();
  // 1) 현재 챕터 단어장 (문맥상 가장 정확)
  const c=rdBookCur&&rdBookCur.chapters[rdChapCur];
  if(c&&c.words){
    let h=c.words.find(x=>x.w===w) || c.words.find(x=>x.r===r)
       || c.words.find(x=>x.w&&w&&(x.w.indexOf(w)>=0||w.indexOf(x.w)>=0));
    if(h) return h.ko;
  }
  // 2) 표기 → 3) 읽기 → 4) 송가나 뗀 표기
  return I.byW.get(w) || (r&&I.byR.get(r)) || I.byW.get(rdTrim(w)) || null;
}
// 뜻을 못 찾았을 때: 한자를 낱글자로 쪼개 뜻을 붙여 준다 (建築 → 建 세우다 · 築 쌓다)
function rdKanjiHint(w){
  const I=rdBuildIndex();
  const out=[];
  for(const ch of String(w||"")){
    if(!/[一-鿿]/.test(ch)) continue;
    const ko=I.byK.get(ch)||I.byW.get(ch);
    if(ko) out.push(`${ch} ${String(ko).split(/[,，·]/)[0].trim()}`);
  }
  return out.length? out.join(" · ") : "";
}
