function wbSearch(){
  const q=($("wbQ").value||"").trim();
  const box=$("wbRes");
  if(q.length<1){ box.innerHTML=""; return; }
  const have=getWB();
  const hit=[];
  for(const cat in DATA){
    if(cat.startsWith("__")) continue;
    for(const c of (DATA[cat]||[])){
      const f=c.front||"", ko=c.ko||"", rd=c.romaji||"";
      if(f.indexOf(q)<0 && ko.indexOf(q)<0 && rd.indexOf(q)<0) continue;
      if(have[cat+"|"+f]) continue;                 // 이미 담긴 건 빼고
      hit.push({cat,c});
      if(hit.length>=40) break;
    }
    if(hit.length>=40) break;
  }
  // 앞에서부터 일치하는 것을 위로
  hit.sort((a,b)=>((a.c.front||"").indexOf(q)===0?0:1)-((b.c.front||"").indexOf(q)===0?0:1));
  box.innerHTML = hit.length
    ? hit.slice(0,12).map(({cat,c})=>{
        const nm=(NAV_NAME[cat]||cat);
        const mean=c.sent ? (c.exk||c.ko||"") : (c.ko||"");
        return `<button onclick="wbAddManual('${cat}','${String(c.front).replace(/'/g,"\\'")}')">
          <span class="f">${c.front}</span><span class="m">${mean}</span><span class="p">＋ ${nm}</span></button>`;
      }).join("")
    : `<div class="none">"${q}" — 찾는 단어가 없어요</div>`;
}
