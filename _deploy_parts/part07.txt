function speakP(t){
  return new Promise(res=>{
    t=String(t||""); if(!t||MUTED) return res();
    let fired=false, guard=0;
    const fin=()=>{ if(!fired){ fired=true; clearTimeout(guard); res(); } };
    // ⚠ 안전장치는 "문장 길이에 비례"해야 한다. 예전엔 9초 고정이라
    //    책의 긴 문장에서 재생 도중에 다음 문장으로 넘어가 "이어 듣기가 끊기는" 증상이 났음.
    const fallbackMs = Math.max(9000, t.length*260 + 4000);
    if(AUDIO_MANIFEST){
      const h=AUDIO_MANIFEST[t]||AUDIO_MANIFEST[t.split("/")[0]];
      if(h){
        try{
          const a=_getAudioEl(); const src="audio/"+h+".mp3";
          if(!a.src || a.src.indexOf(src)===-1){ a.src=src; a.load(); }
          a.onended=fin;
          a.onerror=()=>{ if(!fired) speakWSP(t).then(fin); };
          a.currentTime=0;
          // 실제 길이를 알면 그걸 기준으로(＋여유 1.2초) 안전장치를 다시 건다
          const arm=()=>{ if(fired) return; clearTimeout(guard);
            const ms = (isFinite(a.duration)&&a.duration>0) ? a.duration*1000+1200 : fallbackMs;
            guard=setTimeout(fin,ms); };
          a.onloadedmetadata=arm; a.ondurationchange=arm;
          guard=setTimeout(fin,fallbackMs); arm();
          const pr=a.play(); if(pr&&pr.catch) pr.catch(()=>{ if(!fired) speakWSP(t).then(fin); });
          return;
        }catch(e){}
      }
    }
    speakWSP(t).then(fin); guard=setTimeout(fin,fallbackMs);
  });
}

function readingVariants(r){
  const out=new Set(); const s=String(r||"");

    if(_DAKU[c]) out.add(s.slice(0,i)+_DAKU[c]+s.slice(i+1));

  }
