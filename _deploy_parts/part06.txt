function initPaddle(){
  try{
    if(window.Paddle && !_paddleReady){
      Paddle.Initialize({ token: PADDLE_CLIENT_TOKEN, eventCallback: function(ev){
        if(ev && ev.name==="checkout.completed"){

function computePrice(n){ n=+n||0; const rate = n<20?40 : n<100?30 : n<200?25 : n<300?20 : 15; return Math.round(n*rate*0.5/100)*100; }  // 런칭 50% 할인

function cgGroups(c){
  const pick=(sec,names)=> (NAV[sec]||[]).filter(g=> names.includes("*") || names.includes(g.ko));
  let gs = c.sec ? pick(c.sec, c.grp) : [];
  if(c.sec2) gs = gs.concat(pick(c.sec2, c.grp2));
  (c.more||[]).forEach(([sec,names])=>{ gs = gs.concat(pick(sec,names)); });
  // only / not — 그룹 안에서 팩 단위로 골라낸다 (초급만 / 초급 빼고)
  if(c.only || c.not){
    gs = gs.map(g=>({...g, items:g.items.filter(it=>
          c.only ? c.only.includes(it.code) : !c.not.includes(it.code))}))
           .filter(g=>g.items.length);
  }
  return gs;
}
function cgCount(c){ return cgGroups(c).reduce((n,g)=>n+g.items.length,0); }
function cgFree(c){ return cgGroups(c).some(g=>g.items.some(it=>tierOf(it)!=="paid")); }

async function pickCat(it, toVocab){

  const owned = paid && isOwned(it.code);
  const have = (DATA[it.code]||[]).length;
  if(!paid && !it.ready){ toast("곧 준비됩니다"); return; }            // 무료 준비중
  if(paid && !owned){                                                 // 유료 미보유 → 미리보기 20장
    if(!have){ toast("곧 준비됩니다"); return; }
    previewMode=true;
  } else previewMode=false;
  if(paid && owned && have<=20){                                      // 보유 유료 → 전체 100장 지연 로딩
    const ok = await loadPaidCategory(it);
    if(!ok){ toast("구매 복원(이메일+키)이 필요하거나 서버 응답에 실패했어요"); return; }

  curCat=it;
  $("setupK").textContent=it.ko; $("setupJ").textContent=it.jp;
  const sentPack = !!it.sent;
  $("typeRow").style.display = isKanjiCat() ? "block" : "none";
  [...$("segType").children].forEach(b=>{
    const bm=b.dataset.m;
    b.style.display = sentPack ? ((bm==="meaning"||bm==="listen")?"":"none") : "";
    if(bm==="meaning") b.textContent = sentPack ? "빈칸 채우기" : "뜻 맞히기";
  });
  if(sentPack && (CONF.mode==="reading"||CONF.mode==="write")){
    CONF.mode="meaning";
    [...$("segType").children].forEach(b=>b.classList.toggle("on",b.dataset.m==="meaning"));
  }
  const pv=$("previewNote");
  if(pv){
    pv.style.display = previewMode ? "block" : "none";
    if(previewMode) pv.innerHTML=`🎁 미리보기 <b>10장 무료 체험</b> 중 · 전체 ${it.n}장 잠금해제 <b>${priceOf(it)}</b> <button class="buy-mini" onclick="openBuy(curCat)">구매</button>`;

  const pool = (dailyMode||taMode) ? DAILY_POOL : (wbMode ? (DATA[deck[qi]._cat]||DAILY_POOL) : DATA[curCat.code]);
  let wrongs;
  if(m==="reading" && !dailyMode && isKanjiCat()){
    const vars=shuffle(readingVariants(q.romaji)).slice(0,3).map((v,i)=>({front:"__v"+i, romaji:v, ko:"", speak:""}));
    wrongs = vars.length>=3 ? vars : vars.concat(shuffle(pool.filter(x=>x.front!==q.front && x.romaji!==q.romaji)).slice(0,3-vars.length));
  } else {
    wrongs=shuffle(pool.filter(x=>x.front!==q.front)).slice(0,3);
  }
  const opts=shuffle([q,...wrongs]);
  const box=$("opts");box.className="opts";box.innerHTML="";
  opts.forEach(o=>{const b=document.createElement("button");b.className="opt";b.textContent=optText(o);b.dataset.front=o.front;b.onclick=()=>choose(b,o,q);box.appendChild(b);});
  // 문제가 뜰 때 발음: 읽기 맞히기 모드는 답이 노출되므로 제외, 그 외(뜻모드·카나)는 바로 발음
  if(m==="listen" && !dailyMode){ sched(()=>speak(isSent()? q.exy : q.speak), 300); }
  else if(!(isKanjiCat() && (m==="reading"||m==="write")) && !isSent()){ sched(()=>speak(q.speak), 250); }
  startTimer();
}
function startTimer(){
  if(taMode) return;                 // 타임어택은 전역 60초 타이머가 관리
  clearInterval(timerId);
  const confTime = dailyMode ? 5 : CONF.time;
  if(confTime===0){$("tbar").style.width="100%";$("tlabel").textContent="∞";return;}
  const secs = isSent() ? Math.max(confTime,8) : (qMode()==="listen" ? Math.max(confTime,5) : confTime);   // 문장(cloze)은 읽을 시간 최소 8초 보장
  const total=secs*1000, t0=performance.now();
  $("tbar").style.width="100%";
  timerId=setInterval(()=>{ if(paused){return;}
    const left=Math.max(0,total-(performance.now()-t0));
    $("tbar").style.width=(left/total*100)+"%";
    $("tlabel").textContent="0:0"+Math.ceil(left/1000);
    if(left<=0){clearInterval(timerId);timeUp();}
  },50);
}
function choose(btn,o,q){
  if(answered)return; answered=true; clearInterval(timerId);
  const box=$("opts");box.classList.add("locked");
  const ok = o.front===q.front;
  results.push(ok);
  if(ok){btn.classList.add("correct");correct++; combo++; collectCard(q); bonusDrop(); if(wbMode) wbRemove(q); }
  else{btn.classList.add("wrong");combo=0;misses.push(q); wbAdd(q);[...box.children].forEach(b=>{if(b.dataset.front===q.front)b.classList.add("correct");});}
  [...box.children].forEach(b=>{if(!b.classList.contains("correct")&&!b.classList.contains("wrong"))b.classList.add("dim");});
  if(qMode()==="listen" && !dailyMode){ const g=$("glyph"); g.classList.remove("listen"); g.onclick=null;
    if(isSent()){ g.textContent=q.ex; g.classList.add("cloze"); }
    else { g.textContent=q.front; g.classList.toggle("small",q.front.length>1); } }
  updateComboBadge();
  sfx(ok?"ok":"fail");
  if(taMode){                                   // 타임어택: 즉시 다음 문제 + 점수
    taCount++;
    if(ok){ const gain=100+combo*20; taScore+=gain; taMax=Math.max(taMax,combo); taPop("+"+gain+(combo>=3?" 🔥x"+combo:"")); }
    sched(nextQ, ok ? 250 : 650);
    return;
  }
  const showEx = isKanjiCat() && q.ex;
  if(showEx) showExample(q);
  if(!ok && (CONF.rep|0)>0){ sched(()=>runRepeat(q), 420); return; }   // 틀린 문장 N회 반복 학습
  sched(()=>speak(showEx ? q.exy : q.speak), ok ? 550 : 500); // 효과음 후 발음 (예문 우선)
  schedNext(showEx ? 3200 : 1300);                            // 예문 재생이 잘리지 않게 여유
}
