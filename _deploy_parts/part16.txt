  },80);
  speak(c.speak||c.front);
}
function svPick(i){
  if(svLock) return; svLock=true; clearInterval(svTimer);
  const btns=[...$("svOpts").querySelectorAll("button")];
  const ci=svQ.opts.findIndex(o=>o.ko===svQ.c.ko);
  btns.forEach(b=>b.disabled=true); btns[ci].classList.add("ok");
  if(i!==ci) btns[i].classList.add("no");
  svHit(i===ci);
}
function svHit(ok){
  svLock=true;
  if(ok){ svScore++; try{sfx("ok");}catch(e){} sched(svNext,420); }
  else{
    svLife--; try{sfx("no");}catch(e){} svRenderLife();
    if(svLife<=0) sched(svOver,700); else sched(svNext,900);
  }
}
function svOver(){
  clearInterval(svTimer);
  $("svPlay").style.display="none"; $("svOver").style.display="block";
  const best=svBest(); const nr=svScore>best;
  if(nr) svBestSet(svScore);
  $("svEm").textContent = nr?"🏆":svScore>=20?"🎉":svScore>=10?"💪":"💀";
  $("svSc").textContent=svScore+"문제";
  $("svMs").textContent = nr ? `신기록! 이전 최고 ${best}문제` : `최고 기록 ${best}문제`;
  $("svBest").textContent=svBest();
}
