// ── 채점: 가나 정규화 후 유사도 ──
function spNorm(t){
  let x=String(t||"").replace(/[\s、。「」『』・！？!?,.]/g,"");
  // 가타카나 → 히라가나
  x=x.replace(/[ァ-ヶ]/g,c=>String.fromCharCode(c.charCodeAt(0)-0x60));
  // 장음·촉음 통일(인식기가 흔히 흘림)
  x=x.replace(/ー/g,"").replace(/っ/g,"");
  return x;
}
function spSim(a,b){
  a=spNorm(a); b=spNorm(b);
  if(!a||!b) return 0;
  const m=a.length, n=b.length;
  const d=Array.from({length:m+1},(_,i)=>[i,...Array(n).fill(0)]);
  for(let j=0;j<=n;j++) d[0][j]=j;
  for(let i=1;i<=m;i++) for(let j=1;j<=n;j++)
    d[i][j]=Math.min(d[i-1][j]+1, d[i][j-1]+1, d[i-1][j-1]+(a[i-1]===b[j-1]?0:1));
  return Math.max(0, 1 - d[m][n]/Math.max(m,n));
}

  const pass=score>=70;

function spDiffHtml(heard, target){
  const h=spNorm(heard), t=spNorm(target);
  let out="", hi=0;
  for(const ch of t){
    if(hi<h.length && h[hi]===ch){ out+=`<span class="hit">${ch}</span>`; hi++; }
    else out+=`<span class="miss">${ch}</span>`;
  }
  return out;
}
