function mqShow(){
  if(mqI>=mqList.length){
    $("mqPic").textContent = mqOk===mqList.length ? "🏆" : "🎉";
    $("mqSay").innerHTML=`${mqList.length}문제 중 <b>${mqOk}</b>개 정답!`;
    $("mqOpts").innerHTML=`<button style="grid-column:1/-1;font-size:16px;font-family:inherit" onclick="mnQuizStart()">다시 하기</button>`;
    $("mqFb").textContent=""; $("mqPos").textContent="완료";
    return;
  }
  const q=mqList[mqI];
  $("mqPic").textContent=q[2];
  $("mqSay").innerHTML=`<span style="font-size:17px">이 그림은 <b>${q[3]}</b> — 어떤 글자일까요?`;
  $("mqPos").textContent=`${mqI+1} / ${mqList.length}`;
  $("mqFb").textContent="";
  const wrong=shuffle(mnData().filter(x=>x[0]!==q[0])).slice(0,3);
  const opts=shuffle([q,...wrong]);
  $("mqOpts").innerHTML=opts.map(o=>`<button onclick="mqPick('${o[0]}','${q[0]}',this)">${o[0]}</button>`).join("");
}
function mqPick(c,a,el){
  if(el.parentNode.dataset.done) return;
  speak(c);
  if(c===a){
    el.parentNode.dataset.done=1;
    el.classList.add("ok"); mqOk++;
    $("mqFb").innerHTML=`정답! <b>${a}</b> — ${mnData().find(x=>x[0]===a)[4]}`;
    setTimeout(()=>{ mqI++; $("mqOpts").dataset.done=""; mqShow(); },1400);
  }else{
    el.classList.add("no");
    $("mqFb").textContent=`이건 「${c}」예요. 다시 골라 보세요!`;
  }
}
