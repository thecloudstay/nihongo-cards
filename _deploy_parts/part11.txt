// 마크업을 깨지 않고 문절 단위로 자른다. 구두점 뒤 · 조사 뒤(다음이 한자/가타카나일 때)
function rdChunks(t){
  const parts=[]; let buf="", i=0;
  const PART="はがをにへでとものやかなねよからまでよりこそさえしか";
  while(i<t.length){
    if(t[i]==="{"){ const j=t.indexOf("}",i); const seg=t.slice(i,(j<0?t.length:j+1)); buf+=seg; i+=seg.length; continue; }
    const ch=t[i]; buf+=ch; i++;
    if("、。！？「」』）".includes(ch)){ parts.push(buf); buf=""; continue; }
    // 조사 뒤 + 다음 글자가 한자/가타카나면 단어 중간일 확률이 낮다
    if(PART.includes(ch)){
      const nx=t[i]==="{" ? t[i+1] : t[i];
      if(nx && /[一-鿿ァ-ヺ]/.test(nx) && rdPlain(buf).length>=3){ parts.push(buf); buf=""; }
    }
  }
  if(buf) parts.push(buf);
  // 너무 짧은 조각은 앞과 합침
  const out=[];
  for(const p of parts){
    const n=rdMora(p);
    if(out.length && n<=2) out[out.length-1]+=p; else out.push(p);
  }
  return out;
}
// 모라 수 — 작은 ゃゅょ는 앞 글자와 한 몸, ー·っ은 1모라. 구두점은 쉼으로 가중.
function rdMora(seg){
  const r=rdReading(seg).replace(/\{|\}/g,"");
  let n=0;
  for(const c of r){
    if("ゃゅょャュョ".includes(c)) continue;
    if("、".includes(c)){ n+=2; continue; }
    if("。！？".includes(c)){ n+=4; continue; }
    if("「」『』（）".includes(c)) continue;
    n++;
  }
  return Math.max(1,n);
}
